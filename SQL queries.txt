CREATE VIEW pitching_stats AS
SELECT
    p.pitcher_id,

    /* Outs → used for IP */
    SUM(p.outs) AS total_outs,

    /* Innings Pitched */
    SUM(p.outs) / 3.0 AS IP,

    /* Hits allowed */
    SUM(p.single_hit + p.double_hit + p.triple_hit + p.homerun) AS Hits,

    /* Runs allowed */
    COUNT(pr.runner_id) AS Runs,

    /* Strikeouts */
    SUM(p.strikeout) AS K,

    /* Walks */
    SUM(p.walk) AS BB,

    /* RA (Runs * 3 / IP) */
    CASE
        WHEN SUM(p.outs) = 0 THEN 0
        ELSE (COUNT(pr.runner_id) * 3.0) / (SUM(p.outs) / 3.0)
    END AS RA,

    /* WHIP = (BB + Hits) / IP */
    CASE
        WHEN SUM(p.outs) = 0 THEN 0
        ELSE
            (SUM(p.walk) + SUM(p.single_hit + p.double_hit + p.triple_hit + p.homerun))
            / (SUM(p.outs) / 3.0)
    END AS WHIP

FROM Play p

LEFT JOIN Play_Run pr
    ON pr.play_id = p.play_id

GROUP BY p.pitcher_id;

CREATE VIEW hitting_stats AS
SELECT
    p.batter_id,

    /* Plate Appearances */
    COUNT(*) AS PA,

    /* At Bats (exclude walks) */
    SUM(CASE WHEN p.walk = 0 THEN 1 ELSE 0 END) AS AB,

    /* Hits */
    SUM(p.single_hit + p.double_hit + p.triple_hit + p.homerun) AS Hits,

    SUM(p.single_hit) AS Singles,
    SUM(p.double_hit) AS Doubles,
    SUM(p.triple_hit) AS Triples,
    SUM(p.homerun) AS Homeruns,

    /* Strikeouts */
    SUM(p.strikeout) AS SO,

    /* Walks */
    SUM(p.walk) AS BB,

    /* Runs (from Play_Run) */
    COUNT(pr.runner_id) AS Runs,

    /* Batting Average */
    CASE
        WHEN SUM(CASE WHEN p.walk = 0 THEN 1 ELSE 0 END) = 0 THEN 0
        ELSE
            SUM(p.single_hit + p.double_hit + p.triple_hit + p.homerun) /
            SUM(CASE WHEN p.walk = 0 THEN 1 ELSE 0 END)
    END AS AVG

FROM Play p

LEFT JOIN Play_Run pr
    ON pr.runner_id = p.batter_id

GROUP BY p.batter_id;

CREATE VIEW hitting_stats AS
SELECT
    pl.batter_id AS batter_id,

    COUNT(pl.play_id) AS PA,

    SUM(CASE WHEN COALESCE(pl.walk, 0) = 0 THEN 1 ELSE 0 END) AS AB,

    SUM(
        COALESCE(pl.single_hit, 0) +
        COALESCE(pl.double_hit, 0) +
        COALESCE(pl.triple_hit, 0) +
        COALESCE(pl.homerun, 0)
    ) AS Hits,

    SUM(COALESCE(pl.single_hit, 0)) AS Singles,
    SUM(COALESCE(pl.double_hit, 0)) AS Doubles,
    SUM(COALESCE(pl.triple_hit, 0)) AS Triples,
    SUM(COALESCE(pl.homerun, 0)) AS Homeruns,

    SUM(COALESCE(pl.strikeout, 0)) AS SO,
    SUM(COALESCE(pl.walk, 0)) AS BB,

    COUNT(pr.runner_id) AS Runs,

    CASE
        WHEN COUNT(pl.play_id) = 0 THEN 0
        ELSE
            SUM(
                COALESCE(pl.single_hit, 0) +
                COALESCE(pl.double_hit, 0) +
                COALESCE(pl.triple_hit, 0) +
                COALESCE(pl.homerun, 0)
            ) / COUNT(pl.play_id)
    END AS AVG

FROM Play pl
LEFT JOIN Play_Run pr
    ON pl.play_id = pr.play_id
    AND pr.runner_id = pl.batter_id

WHERE pl.batter_id IS NOT NULL
  AND pl.batter_id != -1

GROUP BY pl.batter_id;

CREATE VIEW pitching_stats AS
SELECT
    pl.pitcher_id AS pitcher_id,

    SUM(pl.outs) AS total_outs,

    SUM(pl.outs) / 3.0 AS IP,

    SUM(
        COALESCE(pl.single_hit, 0) +
        COALESCE(pl.double_hit, 0) +
        COALESCE(pl.triple_hit, 0) +
        COALESCE(pl.homerun, 0)
    ) AS Hits,

    COUNT(pr.runner_id) AS Runs,

    SUM(COALESCE(pl.strikeout, 0)) AS K,

    SUM(COALESCE(pl.walk, 0)) AS BB,

    CASE
        WHEN SUM(pl.outs) = 0 THEN 0
        ELSE (COUNT(pr.runner_id) * 3.0) / SUM(pl.outs)
    END AS RA,

    CASE
        WHEN SUM(pl.outs) = 0 THEN 0
        ELSE
            (
                SUM(COALESCE(pl.walk, 0)) +
                SUM(
                    COALESCE(pl.single_hit, 0) +
                    COALESCE(pl.double_hit, 0) +
                    COALESCE(pl.triple_hit, 0) +
                    COALESCE(pl.homerun, 0)
                )
            ) / (SUM(pl.outs) / 3.0)
    END AS WHIP

FROM Play pl
LEFT JOIN Play_Run pr
    ON pl.play_id = pr.play_id

WHERE pl.pitcher_id IS NOT NULL
  AND pl.pitcher_id != -1

GROUP BY pl.pitcher_id;
